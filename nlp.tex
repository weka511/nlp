\newglossaryentry{gls:softmax}{
	name={Softmax},
	description={
		A function used to convert a tuple of real numbers, positive or negative, to probabilities\cite[Section 6.2.2.3]{goodfellow2016deep}.
		\begin{align*}
			\sigma(\vec{z})_k =& \frac{e^{z_k}}{\sum_{k^{\prime}} e^{z_{k^{\prime}}}} \numberthis \label{eq:glossary:softmax}
		\end{align*}
}}

\newglossaryentry{gls:crossentropy}{
	name={Cross Entropy},
	description={The cross-entropy of the distribution $p$ relative to a distribution $q$  over a given set is defined as follows:\cite[Section 3.13]{goodfellow2016deep}
	\begin{align*}
		H(q,p) =& \mathbb{E}_q[\log{p}]\\
		=& \sum_k p_k \log{q_k} \numberthis \label{eq:glossary:crossentropy}
	\end{align*}
	}
}

\section{Continuous Bag of Words}
In this section I derive the equations needed to implement \gls{gls:cbow}\cite{mikolov2013efficient}.

\subsection{Model}

\begin{align*}
	x_i =& \frac{1} {m} \sum_j w_i^{(j)} \numberthis \label{eq:def:x}\\
	y_j =& \sum_i P_{ji} x_i  \numberthis \label{eq:def:y}\\
	z_k =& \sum_j H_{kj} y_j \numberthis \label{eq:def:z}\\\
	\Lambda[P,H,Label] =& -\sum_k  \log {p_k} Label_k \text {, the \gls{gls:crossentropy}\cite{goodfellow2016deep}}\\
	=& -\log {p_m} \text{, if the label is a one-hot vector $e_m$}
\end{align*}
where the $p_k=\sigma(\vec{z})_k$ are calculated using the \gls{gls:softmax} function \eqref{eq:glossary:softmax}.

\begin{align*}
	p_k =& \frac{e^{z_k}}{\sum_{k^{\prime}} e^{z_{k^{\prime}}}} \text{, whence}\\
	\log{p_k} =& z_k - \ln \sum_{k^{\prime}} e^{z_{k^{\prime}}} \text{, and}\\
	\Lambda[P,H,e_m] =& -z_m +  \ln \sum_{k} e^{z_{k}}  \numberthis \label{eq:def:lambda}\
\end{align*}

\subsection{Calculating the Gradient}

I will use the chain rule to calculate $\frac{\partial{\Lambda}}{\partial H_{ij}}$ and $\frac{\partial{\Lambda}}{\partial P_{ij}}$.
\begin{align*}
	\frac{\partial{\Lambda}}{\partial H_{ij}} =& \sum_k \frac{\partial{\Lambda}}{\partial z_k}\frac{\partial z_k}{\partial H_{ij}} \numberthis \label{eq:Lambda:H}\\
	\frac{\partial{\Lambda}}{\partial P_{ij}} =&\sum_{k,j^{\prime}} \frac{\partial{\Lambda}}{\partial z_k} \frac{\partial z_k}{\partial y_{j^{\prime}}} \frac{\partial y_{j^{\prime}}}{\partial P_{ij}} \numberthis \label{eq:Lambda:P}
\end{align*}

Differentiating \eqref{eq:def:lambda} and \eqref{eq:def:z}:
\begin{align*}
	\frac{\partial \Lambda}{\partial z_k} =& \frac{\partial [-z_m +  \ln \sum_{k^{\prime}} e^{z_{k^{\prime}}}]}{\partial z_k}\\
	=& - \delta_{mk} + \frac{\partial [\ln \sum_{k^{\prime}} e^{z_{k^{\prime}}}]}{\partial z_k}\\
	=& - \delta_{mk} + \frac{e^{z_k}}{\sum_{k^{\prime}} e^{z_{k^{\prime}}}} \numberthis \label{eq:Lambda:z}\\
	\frac{\partial z_k}{\partial H_{ij}} =& \frac{ \partial \sum_{j^{\prime}} H_{k{j^{\prime}}} y_{j^{\prime}}}{\partial H_{ij}}\\
	=& \delta_{ik} \sum_{j^{\prime}} \delta_{j{j^{\prime}}} y_{j^{\prime}}\\
	=& \delta_{ik} y_j \numberthis \label{eq:z:H}\\
\end{align*}

Combining \eqref{eq:Lambda:H}, \eqref{eq:Lambda:z}, and \eqref{eq:z:H}:
\begin{align*}
	\frac{\partial{\Lambda}}{\partial H_{ij}} =& \sum_k  [-\delta_{mk} + \frac{e^{z_k}}{\sum_{k^{\prime}} e^{z_{k^{\prime}}}} ] \delta_{ik} y_j\\
	=& [-\delta_{mi} + \frac{e^{z_i}}{\sum_{k^{\prime}} e^{z_{k^{\prime}}}} ]  y_j   \numberthis \label{eq:Lambda:HH}
\end{align*}
Differentiating \eqref{eq:def:y} :
\begin{align*}
	\frac{\partial y_k}{\partial P_{ij}} =&\frac{\partial \sum_{i^{\prime}} P_{k{i^{\prime}}} x_{i^{\prime}}}{\partial P_{ij}}\\
	=& \delta_{ik} \delta_{j{i^{\prime}}} x_{{i^{\prime}}}\\
	=& \delta_{ik} x_j
\end{align*}

From \eqref{eq:Lambda:P}, \eqref{eq:Lambda:z}, \eqref{eq:def:z}, and \eqref{eq:def:y}.
\begin{align*}
	\frac{\partial{\Lambda}}{\partial P_{ij}} =&\sum_{k,j^{\prime}} [-\delta_{mk} + \frac{e^{z_k}}{\sum_{k^{\prime}} e^{z_{k^{\prime}}}}] H_{k{j^{\prime}}} \delta_{ij^{\prime}} x_j\\
	=&\sum_{k,} [-\delta_{mk} + \frac{e^{z_k}}{\sum_{k^{\prime}} e^{z_{k^{\prime}}}}] H_{ki}  x_j \numberthis \label{eq:Lambda:PP}
\end{align*}
